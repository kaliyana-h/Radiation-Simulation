# ============================================================
# Lunar Habitat — Quonset Hut (half-cylinder on surface)
# ============================================================
#
# A half-cylindrical shell lying on the lunar surface.
# The flat opening faces downward (Z=0, habitat floor).
# The arch rises upward to Z = HabitatOuterRadius.
# The cylinder axis runs along Y (RotX=90 rotates from Z to Y).
#
# Scorer design avoids any overlap with the curved arch wall:
#   OuterBox — thin box well ABOVE the arch top (Z > OuterRadius)
#              captures all beam particles before they hit the arch
#   InnerBox — thin box at Z=1 cm (just above floor, inside arch)
#              all X within inner radius, so no overlap with wall
#   Phantom  — water sphere at arch mid-height (crew proxy)
#
# GUI shielding metrics:
#   Transmission = fluence_inside / fluence_outside
#   Crew dose    = PhantomDose (Gy)
# ============================================================


# ------------------------------------------------------------
# Top-level settings
# ------------------------------------------------------------
i:Ts/Seed            = 1
i:Ts/NumberOfThreads = 1
b:Ts/PauseBeforeQuit = "true"


# ------------------------------------------------------------
# Physics
# ------------------------------------------------------------
s:Ph/Default/Type = "FTFP_BERT_HP"   # REDMoon neutron transport (see lunar_environment.txt)


# ------------------------------------------------------------
# World volume
# ------------------------------------------------------------
s:Ge/World/Type      = "TsBox"
s:Ge/World/Material  = "Vacuum"
d:Ge/World/HLX       = 1200.0 cm
d:Ge/World/HLY       = 1200.0 cm
d:Ge/World/HLZ       = 1200.0 cm
b:Ge/World/Invisible = "true"


# ============================================================
# GUI-FACING PARAMETERS
# ============================================================
s:Ge/DomeMaterial       = "G4_POLYETHYLENE"
d:Ge/HabitatInnerRadius = 400.0 cm
d:Ge/WallThickness      = 5.0 cm
d:Ge/HabitatHL          = 600.0 cm


# ============================================================
# DERIVED PARAMETERS
# ============================================================
d:Ge/HabitatOuterRadius = Ge/HabitatInnerRadius + Ge/WallThickness cm


# ------------------------------------------------------------
# Half-cylinder arch wall — grey solid
# RotX=90 rotates cylinder axis from Z to Y.
# SPhi=0, DPhi=180 keeps the upper half (arch over Z>0).
# Flat opening sits at Z=0 (ground level).
# ------------------------------------------------------------
iv:Gr/Color/DomeGlass = 4 170 170 170 60

s:Ge/HabitatWall/Type         = "TsCylinder"
s:Ge/HabitatWall/Parent       = "World"
s:Ge/HabitatWall/Material     = Ge/DomeMaterial
d:Ge/HabitatWall/RMin         = Ge/HabitatInnerRadius cm
d:Ge/HabitatWall/RMax         = Ge/HabitatOuterRadius cm
d:Ge/HabitatWall/HL           = Ge/HabitatHL cm
d:Ge/HabitatWall/SPhi         = 0.0 deg
d:Ge/HabitatWall/DPhi         = 180.0 deg
d:Ge/HabitatWall/RotX         = -90.0 deg
d:Ge/HabitatWall/TransX       = 0.0 cm
d:Ge/HabitatWall/TransY       = 0.0 cm
d:Ge/HabitatWall/TransZ       = 0.0 cm
s:Ge/HabitatWall/Color        = "DomeGlass"
s:Ge/HabitatWall/DrawingStyle = "Solid"


# ------------------------------------------------------------
# Outer shell scorer — yellow wireframe half-cylinder outline
# Same orientation and position as HabitatWall, but with larger radius
# to create the visible yellow outline without overlapping the wall.
# ------------------------------------------------------------
s:Ge/OuterShell/Type         = "TsCylinder"
s:Ge/OuterShell/Parent       = "World"
s:Ge/OuterShell/Material     = "G4_AIR"
d:Ge/OuterShell/RMin         = Ge/HabitatOuterRadius + 1.0 cm
d:Ge/OuterShell/RMax         = Ge/HabitatOuterRadius + 3.0 cm
d:Ge/OuterShell/HL           = Ge/HabitatHL cm
d:Ge/OuterShell/SPhi         = 0.0 deg
d:Ge/OuterShell/DPhi         = 180.0 deg
d:Ge/OuterShell/RotX         = -90.0 deg
d:Ge/OuterShell/TransX       = 0.0 cm
d:Ge/OuterShell/TransY       = 0.0 cm
d:Ge/OuterShell/TransZ       = 0.0 cm
s:Ge/OuterShell/Color        = "yellow"
s:Ge/OuterShell/DrawingStyle = "Wireframe"


# ------------------------------------------------------------
# Inner scorer box — just above floor level inside the arch
# At Z=1 cm, any |X| < InnerRadius is safely inside the arch.
# HLX = InnerRadius - 4 = 396 cm ensures no wall contact.
# ------------------------------------------------------------
s:Ge/InnerBox/Type         = "TsBox"
s:Ge/InnerBox/Parent       = "World"
s:Ge/InnerBox/Material     = "G4_AIR"
d:Ge/InnerBox/HLX          = 396.0 cm
d:Ge/InnerBox/HLY          = 596.0 cm
d:Ge/InnerBox/HLZ          = 1.0 cm
d:Ge/InnerBox/TransX       = 0.0 cm
d:Ge/InnerBox/TransY       = 0.0 cm
d:Ge/InnerBox/TransZ       = 2.0 cm
s:Ge/InnerBox/Color        = "green"
s:Ge/InnerBox/DrawingStyle = "Wireframe"


# ------------------------------------------------------------
# Crew phantom — water sphere at arch mid-height
# ------------------------------------------------------------
s:Ge/Phantom/Type         = "TsSphere"
s:Ge/Phantom/Parent       = "World"
s:Ge/Phantom/Material     = "G4_WATER"
d:Ge/Phantom/RMin         = 0.0 cm
d:Ge/Phantom/RMax         = 20.0 cm
d:Ge/Phantom/TransX       = 0.0 cm
d:Ge/Phantom/TransY       = 0.0 cm
d:Ge/Phantom/TransZ       = 200.0 cm
s:Ge/Phantom/Color        = "blue"
s:Ge/Phantom/DrawingStyle = "Solid"


# ------------------------------------------------------------
# Lunar surface environment  (REDMoon 4-layer regolith + HP physics)
# Defined in the shared, self-contained include below. It sets
# Ph/Default/Type = FTFP_BERT_HP and builds RegolithL1/L2/L3.
# Retune the regolith/physics there, once, for all habitats.
# ------------------------------------------------------------
includeFile = lunar_environment.txt


# ============================================================
# GCR primary source  (generated by make_source.py)
# ------------------------------------------------------------
# Isotropic upper-hemisphere field, solid-angle (Lambert) weighted,
# with a force-field solar-minimum GCR proton spectrum.
# Replaces the old 17 monoenergetic converging-proxy sources.
# Regenerate / retune with, e.g.:
#     python3 make_source.py --rings 5 --azimuth 8 --phi 400
# ============================================================

# BeamRadius (sky-dome source distance) and BeamSpot (per-source disc radius,
# sized to cover the whole habitat footprint) are defined INSIDE
# gcr_source.txt: an included file cannot read parameters from its host, so
# they live in the include. Retune there via make_source.py --beam-radius /
# --beam-spot, then regenerate.

includeFile = gcr_source.txt


# ------------------------------------------------------------
# Scorer A: Fluence above the arch (incoming beam reference)
# ------------------------------------------------------------
s:Sc/OutsideWallFluence/Quantity                  = "Fluence"
s:Sc/OutsideWallFluence/Component                 = "OuterShell"
s:Sc/OutsideWallFluence/OutputFile                = "quonset_fluence_outside"
s:Sc/OutsideWallFluence/OutputType                = "csv"
s:Sc/OutsideWallFluence/IfOutputFileAlreadyExists = "Overwrite"


# ------------------------------------------------------------
# Scorer B: Fluence inside arch near floor (post-shielding)
# ------------------------------------------------------------
s:Sc/InsideWallFluence/Quantity                  = "Fluence"
s:Sc/InsideWallFluence/Component                 = "InnerBox"
s:Sc/InsideWallFluence/OutputFile                = "quonset_fluence_inside"
s:Sc/InsideWallFluence/OutputType                = "csv"
s:Sc/InsideWallFluence/IfOutputFileAlreadyExists = "Overwrite"


# ------------------------------------------------------------
# Scorer C: Absorbed dose at crew position
# ------------------------------------------------------------
s:Sc/PhantomDose/Quantity                  = "DoseToMedium"
s:Sc/PhantomDose/Component                 = "Phantom"
s:Sc/PhantomDose/OutputFile                = "quonset_dose"
s:Sc/PhantomDose/OutputType                = "csv"
s:Sc/PhantomDose/IfOutputFileAlreadyExists = "Overwrite"


# ------------------------------------------------------------

# ------------------------------------------------------------
s:Gr/MyView/Type                = "OpenGL"
uv:Gr/MyView/UpVector           = 3 0.0 0.0 1.0
d:Gr/MyView/Theta               = 72.0 deg

b:Gr/MyView/IncludeAxes         = "true"
d:Gr/MyView/AxesSize            = 500.0 cm

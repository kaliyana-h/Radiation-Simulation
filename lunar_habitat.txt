# ============================================================
# Lunar Habitat Shielding Simulation
# ============================================================
#
# Geometry (all centred on the origin):
#
#   - Hemispherical dome shell (upper hemisphere, Z > 0)
#       Inner radius  = HabitatInnerRadius
#       Outer radius  = HabitatInnerRadius + WallThickness
#       The dome opening sits at the Z=0 plane (habitat floor)
#
#   - Three scorer volumes for comparative shielding measurement:
#       OuterShell  — 2 cm air shell just outside the wall
#                     (measures incoming particle flux before shielding)
#       InnerShell  — 2 cm air shell just inside the wall
#                     (measures flux that penetrated the wall)
#       Phantom     — water sphere at HabitatCentreZ
#                     (measures absorbed dose at crew position)
#
#   - GCR proxy: proton beam from +Z, fires in -Z (TOPAS default)
#       Beam radius auto-matched to dome outer face
#
# GUI shielding effectiveness metrics:
#   Transmission     = fluence_inside / fluence_outside  (0–1)
#   Crew dose        = PhantomDose (Gy)
#
# Next steps toward a realistic model:
#   - Replace beam with an isotropic spherical surface source
#   - Add a GCR energy spectrum (e.g. Badhwar-O'Neill model)
#   - Multi-layer walls: set NumWallLayers > 1 and generate
#     additional nested TsSphere shells via the GUI/script layer
#   - Add a floor slab and regolith layer below Z=0
# ============================================================


# ------------------------------------------------------------
# Top-level settings
# ------------------------------------------------------------
i:Ts/Seed            = 1
i:Ts/NumberOfThreads = 1
b:Ts/PauseBeforeQuit = "true"
# Interactive Qt viewer: mouse rotate (left-drag), zoom (scroll/right-drag),
# pan (middle-drag or shift-drag) + a toolbar.
# NOTE on WSL2: Qt segfaults unless you first disable MIT-SHM and force
# software GL in the shell that launches TOPAS:
#     export QT_X11_NO_MITSHM=1
#     export LIBGL_ALWAYS_SOFTWARE=1
# Then set this to "true". Leave "false" for the static OpenGL window.
b:Ts/UseQt           = "true"


# ------------------------------------------------------------
# Physics
# ------------------------------------------------------------
s:Ph/Default/Type = "FTFP_BERT_HP"   # REDMoon neutron transport (see lunar_environment.txt)


# ------------------------------------------------------------
# World volume
# ------------------------------------------------------------
s:Ge/World/Type      = "TsBox"
s:Ge/World/Material  = "Vacuum"
d:Ge/World/HLX       = 1200.0 cm
d:Ge/World/HLY       = 1200.0 cm
d:Ge/World/HLZ       = 1200.0 cm
b:Ge/World/Invisible = "true"


# ============================================================
# GUI-FACING PARAMETERS — edit these to configure the habitat
# ============================================================

# Wall material — swap to compare shielding performance
# Common options: "G4_Al"  "G4_POLYETHYLENE"  "G4_CONCRETE"  "G4_WATER"
s:Ge/DomeMaterial = "G4_POLYETHYLENE"

# Habitat inner radius (cm) — sets the usable interior size
d:Ge/HabitatInnerRadius = 400.0 cm

# Wall thickness per layer (cm)
d:Ge/WallThickness = 5.0 cm

# Number of wall layers — multi-layer support requires the GUI/script
# to generate additional nested TsSphere shells; single layer here
i:Ge/NumWallLayers = 1

# Height of crew phantom inside the dome (cm)
# A value of ~HabitatInnerRadius/2 places it at dome mid-height
d:Ge/HabitatCentreZ = 200.0 cm


# ============================================================
# DERIVED PARAMETERS — computed from GUI values above
# ============================================================
d:Ge/HabitatOuterRadius = Ge/HabitatInnerRadius + Ge/WallThickness cm

# Outer scorer shell bounds (just outside the wall)
d:Ge/OuterScorerRMin = Ge/HabitatOuterRadius + 1.0 cm
d:Ge/OuterScorerRMax = Ge/HabitatOuterRadius + 3.0 cm

# Inner scorer shell bounds (just inside the wall)
d:Ge/InnerScorerRMax = Ge/HabitatInnerRadius - 1.0 cm
d:Ge/InnerScorerRMin = Ge/HabitatInnerRadius - 3.0 cm


# Translucent grey for the dome (RGBA, 0-255) so the converging
# beams that strike the far/back side remain visible through it.
iv:Gr/Color/DomeGlass = 4 170 170 170 60

# Neutral greys for the two scorer-shell wireframes, so they do NOT
# compete with the particle colours (gammas=yellow, neutrons=green).
iv:Gr/Color/ShellOuter = 4 150 150 150 255
iv:Gr/Color/ShellInner = 4 95  95  120 255

# ============================================================

# ------------------------------------------------------------

#     python3 make_source.py --rings 5 --azimuth 8 --phi 400
# ============================================================

includeFile = lunar_environment.txt


# ------------------------------------------------------------
# Visualisation — tuned for a poster image of the secondary cascade
#
# Camera: UpVector +Z stands the dome on the regolith ground plane;
# Theta=80 looks slightly down from near the horizon so the shower
# develops top-to-bottom in frame; Zoom 0.40 pulls back to fit the
# full 3 m regolith column (yellow 0-22 / orange 22-49 / brown
# 49-300 cm) and the dome. Raise toward 1.5 to crop onto the habitat.
#
# Cascade legibility (the important part):
#   * ColorBy "ParticleType" separates species so the secondaries
#     stand out from the red primary protons:
#         proton  = red     (primaries + knock-on protons)
#         neutron = green   (the key penetrating secondary)
#         gamma   = yellow
#         e- / e+ = cyan / magenta
#         pi+/pi- = white,  alpha = blue
#   * IncludeStepPoints marks interaction vertices (dots) where the
#     cascade branches — i.e. where secondaries are created.
#   * Axes off and thicker lines for a clean printable image.
# ------------------------------------------------------------
s:Gr/MyView/Type                = "OpenGL"
uv:Gr/MyView/UpVector           = 3 0.0 0.0 1.0
d:Gr/MyView/Theta               = 78.0 deg
d:Gr/MyView/Phi                 = 0.0 deg
s:Gr/MyView/Projection          = "Perspective"
d:Gr/MyView/PerspectiveAngle    = 30.0 deg
u:Gr/MyView/Zoom                = 0.40
b:Gr/MyView/IncludeAxes         = "false"
b:Gr/MyView/IncludeTrajectories = "true"
b:Gr/MyView/IncludeStepPoints   = "false"
i:Gr/MyView/TrajectoryLineWidth = 2
i:Gr/MyView/WindowSizeX         = 1600
i:Gr/MyView/WindowSizeY         = 1100

# cyan / magenta / white are not predefined TOPAS color names, so define them:
iv:Gr/Color/cyan    = 4 0   230 230 255
iv:Gr/Color/magenta = 4 255 0   230 255
iv:Gr/Color/white   = 4 255 255 255 255

# Hide the soft (<10 MeV) gamma/electron hash so only the energetic
# shower — the shielding-relevant neutrons and protons — is drawn.
# Filters trajectories by INITIAL kinetic energy; affects drawing only,
# not the scorers.
d:Gr/OnlyIncludeParticlesWithInitialKEAbove = 10. MeV

s:Gr/MyView/ColorBy                    = "ParticleType"
sv:Gr/MyView/ColorByParticleTypeNames  = 8 "proton" "neutron" "gamma" "e-" "e+" "pi+" "pi-" "alpha"
sv:Gr/MyView/ColorByParticleTypeColors = 8 "red" "green" "yellow" "cyan" "magenta" "white" "white" "blue"
